/** * Generates an executive Word document (.doc) from filtered active data, * triggers an automatic file download, copies an email summary to the clipboard, * and launches the OS default email client via mailto. */ async function exportReportToWordAndEmail() { const data = await getFilteredReportData(); // 1. Build and download the styled Word document Blob const wordHtml = buildWordDocumentHTML(data); const blob = new Blob(['\ufeff' + wordHtml], { type: 'application/msword' }); const filename = `TaskTracker_Report_${new Date().toISOString().slice(0, 10)}.doc`; const downloadLink = document.createElement('a'); downloadLink.href = URL.createObjectURL(blob); downloadLink.download = filename; document.body.appendChild(downloadLink); downloadLink.click(); document.body.removeChild(downloadLink); URL.revokeObjectURL(downloadLink.href); // 2. Compose executive cover email text const author = savedSettings.appUsername || savedSettings.username || currentUser.email || 'TaskTracker User'; const coverText = `Hi Team, Please find attached the TaskTracker Executive Performance & Activity Report. Summary Overview: • Tasks & Achievements Logged: ${data.tasks.length} • Scheduled Calendar Events: ${data.calendar.length} • Active To-Do Lists: ${data.todos.length} • Knowledge & Reference Notes: ${data.notes.length} The fully formatted Word report ("${filename}") has been saved to your Downloads folder. Please attach it to this email. Best regards, ${author}`; // 3. Copy cover text to clipboard try { await navigator.clipboard.writeText(coverText); } catch (err) { console.warn('Clipboard copy failed:', err); } // 4. Launch mailto client const recipient = activeSettings.recipients ? encodeURIComponent(activeSettings.recipients) : ''; const subject = encodeURIComponent(`TaskTracker Executive Report - ${new Date().toLocaleDateString()}`); const body = encodeURIComponent(coverText); window.location.href = `mailto:${recipient}?subject=${subject}&body=${body}`; alert(`✅ Formatted Word document generated and downloaded ("${filename}").\n\n✉️ Email client opened! The cover note has been copied to your clipboard. Simply attach the downloaded file to your email.`); } /** * Builds standard MSO-HTML markup with embedded Office XML styling rules. * Generates exact visual parity with native Word document elements. */ function buildWordDocumentHTML(data) { const dateStr = new Date().toLocaleDateString('en-US', { year: 'numeric', month: 'long', day: 'numeric' }); const authorStr = escapeHTML(savedSettings.appUsername || savedSettings.username || currentUser.email || 'TaskTracker User'); let html = ` TaskTracker Executive Report
TaskTracker PRO Executive Report
Generated on ${dateStr} | Prepared by ${authorStr}
`; // 1. TASKS & ACHIEVEMENTS MODULE if (data.tasks.length > 0) { html += `
Tasks & Achievements (${data.tasks.length})
`; data.tasks.forEach(t => { const workType = t.workType || calculateWorkType(t.datetime); const typeClass = t.type === 'Achievement' ? 'badge-achievement' : 'badge-task'; const workClass = workType === 'Overtime' ? 'badge-overtime' : workType === 'Holiday' ? 'badge-holiday' : 'badge-regular'; html += ``; }); html += `
Title Type & Classification Date & Time Details & Outcomes
${escapeHTML(t.title)} ${escapeHTML(t.type || 'Task')}
${escapeHTML(workType)} 		${t.datetime ? new Date(t.datetime).toLocaleString() : 'N/A'} ${escapeHTML(t.desc || 'No details provided.').replace(/\n/g, '
')}
`; } // 2. CALENDAR EVENTS MODULE if (data.calendar.length > 0) { html += `
Calendar Events (${data.calendar.length})
`; data.calendar.forEach(c => { html += ``; }); html += `
Event Title Scheduled Date Notes & Context
${escapeHTML(c.title)} ${escapeHTML(c.date)} ${escapeHTML(c.desc || 'N/A').replace(/\n/g, '
')}
`; } // 3. TO-DO LISTS MODULE if (data.todos.length > 0) { html += `
To-Do Lists (${data.todos.length})
`; data.todos.forEach(list => { html += `
📋 ${escapeHTML(list.title)} (Category: ${escapeHTML(list.category || 'General')} | Target: ${list.dueDate || 'None'})
`; (list.items || []).forEach(item => { html += ``; }); html += `
Status Task Item
${item.completed ? '[ X ] Completed' : '[   ] Pending'} ${escapeHTML(item.text)}
`; }); } // 4. NOTES & KNOWLEDGE LOG MODULE if (data.notes.length > 0) { html += `
Notes & Knowledge Log (${data.notes.length})

Title & Tag Recorded Date Content
${escapeHTML(n.title)}
${escapeHTML(n.tag || 'General')}		 ${new Date(n.timestamp).toLocaleDateString()} ${escapeHTML(n.content).replace(/\n/g, '
')}
`; } html += `
TaskTracker PRO Automated Reporting System • Confidential
`; return html; }
